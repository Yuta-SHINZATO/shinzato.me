% ---------------------------------------------------------
% DATED ENTRY
% ---------------------------------------------------------

% Usage:
%
% \datedentry
%   {Main information}
%   {Date}
%   {Secondary line}
%
% The date is always aligned with the FIRST line,
% even when the left-hand entry spans multiple lines.
\newcommand{\datedentry}[3]{%
  \noindent
  \begin{tabularx}{\textwidth}
  {@{}
   >{\raggedright\arraybackslash}X
   >{\raggedleft\arraybackslash}p{35mm}
   @{}}

    \begin{minipage}[t]{\linewidth}
      \vspace{0pt}%
      #1
    \end{minipage}
    &
    \begin{minipage}[t]{\linewidth}
      \vspace{0pt}%
      \raggedleft
      #2
    \end{minipage}

  \end{tabularx}%
  \par

  \if\relax\detokenize{#3}\relax
  \else
    \vspace{1pt}
    \hspace{5mm}#3
    \par
  \fi
}
